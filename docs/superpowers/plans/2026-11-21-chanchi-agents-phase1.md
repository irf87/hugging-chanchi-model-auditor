# chanchi Agents — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship phase 1 of the agents MVP: a blocking local gate with a CI mirror, the structural checks, the `skill-chanchi-*` rename with a safe link transition, `chanchi agents sync`, three agents, the item-authoring and verify skills, and the feature map.

**Architecture:** Agent sources live in `agents/agent-chanchi-<n>/AGENT.md`; `src/chanchi/agents_sync.py` compiles them into committed per-harness wrappers and `--check` detects drift. Every invariant is a pytest test or a ruff rule run by `scripts/check`, which git hooks and GitHub Actions both call.

**Tech Stack:** Python ≥ 3.11 standard library (runtime), pytest and ruff (dev only), uv, git hooks, GitHub Actions.

**Spec:** `docs/designs/chanchi-agents-mvp.md` (Revision 2). Read it before starting any task.

## Global Constraints

- Runtime dependencies stay empty: `[project] dependencies = []`. Dev tools go in `[dependency-groups] dev`.
- Every file, comment, commit message and report is in English.
- Item names: `<item>-chanchi-<name>`, `<item>` ∈ `agent`, `skill`, `eval`, `hook`; `<name>` matches `[a-z0-9]+(-[a-z0-9]+)*`; directory name == frontmatter `name`.
- Never write a private consumer project's name, a home path (`/Users/...`) or an email into a committed file. The repository is public.
- Tests and manual runs use a scratch database: `export CHANCHI_DB="$(mktemp -d)/chanchi.db"`. Never touch `~/.local/share/chanchi/chanchi.db`.
- Executors do not commit, merge, or write chanchi records. They leave changes in their worktree and write a report that lists every verification command with its last output line. The orchestrator commits (Conventional Commits) and merges with `git merge --no-ff`.
- Error messages of new checks state the rule and the command that fixes it.
- `ChanchiError` codes must come from `src/chanchi/errors.py::ERROR_CODES`; add a code there when a new one is needed.

```python
"""Check 5: import layers of src/chanchi (docs/designs/chanchi-agents-mvp.md, Layer 3)."""

import ast
from pathlib import Path

SRC = Path(__file__).resolve().parents[1] / "src" / "chanchi"
LEAVES = {"errors", "timeutil", "dedup", "hfsignals"}
LEAF_MAY_IMPORT = {"errors"}


def chanchi_imports(path: Path) -> set[str]:
    found = set()
    for node in ast.walk(ast.parse(path.read_text(encoding="utf-8"))):
        if isinstance(node, ast.ImportFrom) and node.module:
            if node.module == "chanchi":
                found.update(alias.name for alias in node.names)
            elif node.module.startswith("chanchi."):
                found.add(node.module.split(".")[1])
        elif isinstance(node, ast.Import):
            for alias in node.names:
                if alias.name.startswith("chanchi."):
                    found.add(alias.name.split(".")[1])
    return found


MODULES = {p.stem: chanchi_imports(p) for p in sorted(SRC.glob("*.py"))}


def test_leaf_modules_import_only_errors():
    for leaf in LEAVES:
        extra = MODULES[leaf] - LEAF_MAY_IMPORT
        assert not extra, (
            f"{leaf}.py imports {sorted(extra)}; leaf modules may import only chanchi.errors. "
            "Move the shared code into a leaf or import it from the caller."
        )


def test_only_main_imports_cli():
    offenders = sorted(m for m, deps in MODULES.items() if "cli" in deps and m != "__main__")
    assert not offenders, f"{offenders} import chanchi.cli; only __main__ may. Move the code out of cli.py."


def test_only_cli_imports_render():
    offenders = sorted(m for m, deps in MODULES.items() if "render" in deps and m != "cli")
    assert not offenders, f"{offenders} import chanchi.render; only cli.py renders output."
```

```python
import ast
import sys
import tomllib
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
SRC = ROOT / "src" / "chanchi"


def test_no_runtime_dependencies():
    project = tomllib.loads((ROOT / "pyproject.toml").read_text(encoding="utf-8"))["project"]

        "chanchi has no runtime dependencies. Put dev tools in [dependency-groups] dev, "
        "or write the feature with the standard library."
    )


def test_src_imports_only_stdlib_and_chanchi():
    allowed = set(sys.stdlib_module_names) | {"chanchi", "__future__"}
    bad = []
    for path in sorted(SRC.glob("*.py")):
        for node in ast.walk(ast.parse(path.read_text(encoding="utf-8"))):
            if isinstance(node, ast.Import):
                names = [a.name for a in node.names]
            elif isinstance(node, ast.ImportFrom) and node.module and node.level == 0:
                names = [node.module]
            else:
                continue
            bad += [f"{path.name}: {n}" for n in names if n.split(".")[0] not in allowed]
    assert not bad, f"non-stdlib imports in src/chanchi: {bad}. chanchi's runtime is stdlib only."
```

```python
def _points_into_repo(entry: Path, skills_dir: Path) -> bool:
    """A symlink whose target (resolved or not) lies inside this repository's skills directory."""
    if not entry.is_symlink():
        return False
    raw = Path(os.readlink(entry))
    target = raw if raw.is_absolute() else entry.parent / raw
    try:
        return Path(os.path.normpath(target)).is_relative_to(skills_dir) or entry.resolve().is_relative_to(skills_dir)
    except OSError:
        return False


def _is_legacy(entry: Path, skills_dir: Path, primary: Path) -> bool:
    """A link an older chanchi created: chanchi-<n> pointing into skills/, or a secondary chaining to a
    primary chanchi-<n> that is itself chanchi's (or already gone). Never a foreign file."""
    if not entry.is_symlink() or not entry.name.startswith(LEGACY_PREFIX) or _is_compat(entry):
        return False
    if _points_into_repo(entry, skills_dir):
        return True
    target = primary / entry.name
    chained = os.readlink(entry) == os.path.relpath(target, entry.parent)
    return chained and (not os.path.lexists(target) or _points_into_repo(target, skills_dir))


def _is_compat(entry: Path) -> bool:
    """A compatibility link chanchi-<n> -> skill-chanchi-<n> (relative, same directory)."""
    if not entry.is_symlink() or not entry.name.startswith(LEGACY_PREFIX):
        return False
    return os.readlink(entry) == f"{SKILL_PREFIX}{entry.name[len(LEGACY_PREFIX):]}"
```

```python
"""Compile harness-neutral agent sources into per-harness wrappers (docs/designs/chanchi-agents-mvp.md)."""

from __future__ import annotations

import json
import re
from dataclasses import dataclass
from pathlib import Path

from chanchi.errors import ChanchiError

SCOPES = ("main", "subagent", "both")
PROFILES = ("read-only", "workspace-write", "orchestrator")
NAME = re.compile(r"^agent-chanchi-[a-z0-9]+(-[a-z0-9]+)*$")
WRAPPER_GLOBS = (".claude/agents/agent-chanchi-*.md", ".opencode/agents/agent-chanchi-*.md",
                 ".agents/agents/agent-chanchi-*/agent.md")

# agy tool names, read from the agy 1.5.14 agent definitions on 2026-11-21. Recheck after an agy upgrade
# (skill-chanchi-verify, Helpers).
AGY_READ_TOOLS = ("view_file", "list_dir", "find_by_name", "grep_search", "run_command", "command_status",
                  "send_command_input", "write_to_file", "search_web", "read_url_content")
AGY_EDIT_TOOLS = ("replace_file_content", "multi_replace_file_content")
AGY_ORCHESTRATOR_TOOLS = ("invoke_subagent", "define_subagent", "manage_task")

# Whether the harness preloads skills natively in this launch mode. Verified live in Task 8;
# where False, the generator writes an explicit "load these skills" line instead.
AGY_PRELOAD_VERIFIED = False


@dataclass(frozen=True)
class AgentSource:
    name: str
    description: str
    scope: str
    profile: str
    skills: list[str]
    body: str


def _parse(path: Path) -> AgentSource:
    text = path.read_text(encoding="utf-8")
    if not text.startswith("---\n") or "\n---\n" not in text[4:]:
        raise ChanchiError("invalid_value", f"{path}: missing frontmatter", field="frontmatter",
                           hint="start the file with a --- block holding name, description, scope, profile, skills")
    head, body = text[4:].split("\n---\n", 1)
    fields: dict[str, str] = {}
    for line in head.splitlines():
        if not line.strip():
            continue
        key, sep, value = line.partition(":")
        if not sep:
            raise ChanchiError("invalid_value", f"{path}: bad frontmatter line {line!r}", field="frontmatter",
                               hint="one `key: value` per line; no comments or nested blocks")
        fields[key.strip()] = value.strip()
    for key in ("name", "description", "scope", "profile"):
        if not fields.get(key):
            raise ChanchiError("invalid_value", f"{path}: {key} is required", field=key)
    if not NAME.match(fields["name"]) or fields["name"] != path.parent.name:
        raise ChanchiError("invalid_value", f"{path}: name must be agent-chanchi-<name> and equal the directory",
                           field="name")
    if fields["scope"] not in SCOPES:
        raise ChanchiError("invalid_value", f"{path}: unknown scope", field="scope", valid_values=SCOPES)
    if fields["profile"] not in PROFILES:
        raise ChanchiError("invalid_value", f"{path}: unknown profile", field="profile", valid_values=PROFILES)
    raw_skills = fields.get("skills", "[]")
    if not (raw_skills.startswith("[") and raw_skills.endswith("]")):
        raise ChanchiError("invalid_value", f"{path}: skills must be [a, b]", field="skills")
    skills = [s.strip() for s in raw_skills[1:-1].split(",") if s.strip()]
    return AgentSource(fields["name"], fields["description"], fields["scope"], fields["profile"], skills,
                       body.strip() + "\n")


def load_sources(agents_dir: Path) -> list[AgentSource]:
    if not agents_dir.is_dir():
        return []
    return [_parse(d / "AGENT.md") for d in sorted(agents_dir.iterdir()) if d.is_dir()]


def _header(source: AgentSource) -> str:
    return f"# generated by chanchi agents sync from agents/{source.name}/AGENT.md; do not edit"


def _skills_line(source: AgentSource) -> str:
    return f"Before starting, load these skills: {', '.join(source.skills)}." if source.skills else ""


def _yaml_list(key: str, values) -> list[str]:
    return [f"{key}:", *[f"  - {v}" for v in values]] if values else []


def _indent(text: str) -> str:
    return "\n".join(f"  {line}" if line else "" for line in text.rstrip("\n").splitlines())


def _doc(head: list[str], body: str) -> str:
    return "---\n" + "\n".join(head) + "\n---\n" + body


def _claude(s: AgentSource) -> str:
    head = [_header(s), f"name: {s.name}", f"description: {json.dumps(s.description)}"]
    if s.profile == "read-only":
        head.append("disallowedTools: Edit, NotebookEdit")
    if s.scope == "main":
        prompt = s.body + (f"\n{_skills_line(s)}\n" if s.skills else "")
        return _doc(head + ["initialPrompt: |", _indent(prompt)], "")
    return _doc(head + _yaml_list("skills", s.skills), "\n" + s.body)


def _opencode(s: AgentSource) -> str:
    mode = {"main": "primary", "subagent": "subagent", "both": "all"}[s.scope]
    head = [_header(s), f"description: {json.dumps(s.description)}", f"mode: {mode}"]
    if s.profile == "read-only":
        head += ["permission:", "  edit: deny", "  write: deny"]
    line = _skills_line(s)
    return _doc(head, "\n" + s.body + (f"\n{line}\n" if line else ""))


def _agy(s: AgentSource) -> str:
    tools = list(AGY_READ_TOOLS)
    if s.profile != "read-only":
        tools += AGY_EDIT_TOOLS
    if s.profile == "orchestrator":
        tools += AGY_ORCHESTRATOR_TOOLS
    head = [_header(s), f"name: {s.name}", f"description: {json.dumps(s.description)}",
            f"mainAgent: {'false' if s.scope == 'subagent' else 'true'}",
            f"subagent: {'false' if s.scope == 'main' else 'true'}", *_yaml_list("tools", tools)]
    line = ""
    if AGY_PRELOAD_VERIFIED:
        head += _yaml_list("preloadSkills", s.skills)
    else:
        line = _skills_line(s)
    return _doc(head, "\n" + s.body + (f"\n{line}\n" if line else ""))


def render(source: AgentSource) -> dict[str, str]:
    return {
        f".claude/agents/{source.name}.md": _claude(source),
        f".opencode/agents/{source.name}.md": _opencode(source),
        f".agents/agents/{source.name}/agent.md": _agy(source),
    }


def sync(root: Path, *, check: bool = False) -> dict:
    root = Path(root).resolve()
    sources = load_sources(root / "agents")
    wanted: dict[str, str] = {}
    for source in sources:
        wanted.update(render(source))
    present = {str(p.relative_to(root)) for pattern in WRAPPER_GLOBS for p in root.glob(pattern)}
    stale = sorted(rel for rel, text in wanted.items()
                   if not (root / rel).is_file() or (root / rel).read_text(encoding="utf-8") != text)
    orphans = sorted(present - set(wanted))
    unchanged = sorted(set(wanted) - set(stale))
    if check:
        if stale or orphans:
            raise ChanchiError("stale_wrappers", f"generated agent wrappers are out of date: {stale + orphans}",
                               field="path", hint="run `chanchi agents sync` and commit the result; never edit "
                               "wrappers by hand, edit agents/<name>/AGENT.md", details={"stale": stale + orphans})
        return {"root": str(root), "agents": [s.name for s in sources], "written": [], "removed": [],
                "unchanged": unchanged}
    for rel in stale:
        (root / rel).parent.mkdir(parents=True, exist_ok=True)
        (root / rel).write_text(wanted[rel], encoding="utf-8")
    for rel in orphans:
        (root / rel).unlink()
        if rel.startswith(".agents/agents/") and not any((root / rel).parent.iterdir()):
            (root / rel).parent.rmdir()
    return {"root": str(root), "agents": [s.name for s in sources], "written": stale, "removed": orphans,
            "unchanged": unchanged}
```

```markdown
# chanchi — item authoring

An item is agent setup a harness loads or runs. Design: `docs/designs/chanchi-agents-mvp.md`.

## Names

`<item>-chanchi-<name>`, kebab-case, where `<item>` is `agent`, `skill`, `eval` or `hook`. The
directory name and the frontmatter `name` are the same string. `tests/test_items.py` enforces
this and fails when an operational file names an item that does not exist.

## Layout

| Item | Source | Notes |
|---|---|---|
| agent | `agents/agent-chanchi-<n>/AGENT.md` | wrappers are generated; never edit `.claude/agents`, `.opencode/agents` or `.agents/agents` by hand |
| skill | `skills/skill-chanchi-<n>/SKILL.md`, optional `references/` | linked into projects by `chanchi skills link` |
| eval | `evals/eval-chanchi-<item>/` | phase 2: from then on every new skill ships with at least one eval case |
| hook | `hooks/hook-chanchi-<n>/` | a harness hook; git hooks are not items |

## Agent source

Frontmatter: `name`, `description` (what it does and what it never does), `scope`
(`main`, `subagent`, `both`), `profile` (`read-only`, `workspace-write`, `orchestrator`),
`skills: [a, b]`. One `key: value` per line, no comments. The body is the role, harness neutral.
Never put a model in it: the orchestrator picks models with `chanchi recommend`.

After editing a source, run `chanchi agents sync` (inside this repository: `uv run chanchi agents sync`), then commit
the source and the wrappers.

## Rule or check?

Before adding a sentence that tells agents what they must do, ask whether a test, a ruff rule, a
CLI check or a hook can enforce it. If one can, write the check (its failure message states the
rule and the fix) and do not add the sentence. Text is for judgment; checks are for invariants.

## Size

`AGENTS.md` ≤ 150 lines, each `SKILL.md` and `AGENT.md` ≤ 300 (`tests/test_rules_budget.py`).
Move detail to `docs/` or `references/` and link to it.

## Last step

Run `scripts/check`. It must pass before the change is handed off.
```

```python
"""Check 3: every CLI command is in FEATURE_MAP.md and every covering test exists."""

import argparse
import re
from pathlib import Path

from chanchi import cli

ROOT = Path(__file__).resolve().parents[1]
MAP = ROOT / "FEATURE_MAP.md"
COVERED = re.compile(r"^Covered by: (.+)$", re.MULTILINE)


def _subparsers(parser):
    for action in parser._actions:
        if isinstance(action, argparse._SubParsersAction):
            return action.choices
    return {}


def commands() -> list[str]:
    found = []
    for group, sub in _subparsers(cli.build_parser()).items():
        leaves = _subparsers(sub)
        found += [f"{group} {name}" for name in leaves] if leaves else [group]
    return sorted(found)


def test_every_command_has_an_entry():
    text = MAP.read_text(encoding="utf-8")
    missing = [c for c in commands() if f"`{c}`" not in text]
    assert not missing, (
        f"FEATURE_MAP.md has no entry for {missing}. Add each under its group's Sub-features, "
        "with how to reach it, how to drive it, the evidence and the covering tests."
    )


def test_covering_tests_exist():
    text = MAP.read_text(encoding="utf-8")
    entries = COVERED.findall(text)
    assert entries, "FEATURE_MAP.md lists no `Covered by:` lines"
    missing = [f for line in entries for f in (s.strip() for s in line.split(",")) if not (ROOT / f).is_file()]
    assert not missing, f"FEATURE_MAP.md names test files that do not exist: {missing}"
```
